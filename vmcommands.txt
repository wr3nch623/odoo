sudo sysctl -w net.ipv4.ip_forward=1

sudo iptables -t nat -A POSTROUTING \
    -s 192.168.56.0/24 \
    -o wlp4s0 \
    -j MASQUERADE


sudo iptables -A FORWARD \
    -s 192.168.56.0/24 \
    -o wlp4s0 \
    -j ACCEPT

sudo iptables -A FORWARD \
    -d 192.168.56.0/24 \
    -i wlp4s0 \
    -m conntrack --ctstate RELATED,ESTABLISHED \
    -j ACCEPT
